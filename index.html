var INV_HEADERS = ['id','logo','name','mrp','brand','category','unit','stock','loose','supplierAgency','status','note'];
var RET_HEADERS = ['returnId','skuId','productName','brand','mrp','prate','amt','unit','retStock','retLoose','reason','remarks','supplierAgency','holdRack','returnedDate','pickupDate','pickupNote','status'];

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var invSheet = getOrCreateSheet_(ss, 'StockList', INV_HEADERS);
    var retSheet = getOrCreateSheet_(ss, 'ReturnList', RET_HEADERS);
    var payload = {
      status: 'ok',
      updatedAt: new Date().toISOString(),
      inventory: sheetToObjects_(invSheet, INV_HEADERS),
      returnLogs: sheetToObjects_(retSheet, RET_HEADERS)
    };
    return jsonResponse_(payload);
  } catch (err) {
    return jsonResponse_({ status: 'error', message: String(err) });
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    var data = JSON.parse(raw);

    if (Array.isArray(data.inventory)) {
      var invSheet = getOrCreateSheet_(ss, 'StockList', INV_HEADERS);
      objectsToSheet_(invSheet, INV_HEADERS, data.inventory);
    }
    if (Array.isArray(data.returnLogs)) {
      var retSheet = getOrCreateSheet_(ss, 'ReturnList', RET_HEADERS);
      objectsToSheet_(retSheet, RET_HEADERS, data.returnLogs);
    }
    return jsonResponse_({ status: 'saved', updatedAt: new Date().toISOString() });
  } catch (err) {
    return jsonResponse_({ status: 'error', message: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet_(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#f1f5f9');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function sheetToObjects_(sheet, expectedHeaders) {
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  var headers = rows[0].map(function(h) { return String(h).trim(); });
  var tz = Session.getScriptTimeZone() || 'Asia/Kolkata';

  return rows.slice(1).filter(function(r) {
    return r.some(function(cell) { return cell !== '' && cell !== null; });
  }).map(function(r) {
    var obj = {};
    expectedHeaders.forEach(function(h) { obj[h] = ''; });
    headers.forEach(function(h, i) {
      if (!h) return;
      var val = r[i];
      if (val instanceof Date && !isNaN(val.getTime())) {
        val = Utilities.formatDate(val, tz, 'yyyy-MM-dd');
      }
      obj[h] = val;
    });
    if ('mrp' in obj) obj.mrp = Number(obj.mrp) || 0;
    if ('prate' in obj) obj.prate = Number(obj.prate) || 0;
    if ('amt' in obj) obj.amt = Number(obj.amt) || 0;
    if ('stock' in obj) obj.stock = Math.max(0, parseInt(obj.stock, 10) || 0);
    if ('loose' in obj) obj.loose = Math.max(0, parseInt(obj.loose, 10) || 0);
    if ('retStock' in obj) obj.retStock = Math.max(0, parseInt(obj.retStock, 10) || 0);
    if ('retLoose' in obj) obj.retLoose = Math.max(0, parseInt(obj.retLoose, 10) || 0);
    return obj;
  });
}

function objectsToSheet_(sheet, headers, list) {
  sheet.clearContents();
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#f1f5f9');
  sheet.setFrozenRows(1);
  if (!list || list.length === 0) return;

  var rows = list.map(function(item) {
    return headers.map(function(h) {
      var val = item[h];
      return (val !== undefined && val !== null) ? val : '';
    });
  });
  var range = sheet.getRange(2, 1, rows.length, headers.length);
  range.setValues(rows);
}
